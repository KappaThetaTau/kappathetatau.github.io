Theta Tau is the nation’s oldest and largest co-ed engineering fraternity. Made up of engineers with diverse interests, Theta Tau brings together some of the best and brightest from the Grainger College of Engineering who strive to develop skills both inside and outside of the engineering workplace. With professional, philanthropic, and social emphases, the fraternity offers its members a wide range of events throughout the year. 

Please register for rush using the link below! You’ll be added to our email list and will receive important information about events.

**Register for Rush:** [Click here to register!](https://docs.google.com/forms/d/e/1FAIpQLSe36sJkXeySndcAOX8l2VPifzXhT8wRFQ_U0jiLJtvVoM5tmw/viewform?usp=header)

<script>
const carouselSlide = document.querySelector('.carousel-slide');
const carouselItems = document.querySelectorAll('.carousel-item');
const prevBtn = document.querySelector('.prev');
const nextBtn = document.querySelector('.next');
let counter = 0;
let isAutoPlay = true;

// Always measure fresh - never cache this, so stale/zero widths from
// early page-load timing can never break the math again.
function slideWidth() {
  return carouselItems[0].getBoundingClientRect().width;
}

function moveCarousel(instant) {
  carouselSlide.style.transition = instant ? 'none' : 'transform 0.5s ease-in-out';
  // Force a reflow so the browser actually applies "none" before the
  // transform changes, instead of batching both style changes together.
  void carouselSlide.offsetHeight;
  carouselSlide.style.transform = 'translateX(' + (-slideWidth() * counter) + 'px)';
}

function nextSlide() {
  if (counter >= carouselItems.length - 1) {
    counter = 0;
    setTimeout(() => {
      moveCarousel(true);
      isAutoPlay = true;
    }, 500);
  } else {
    counter++;
    moveCarousel(false);
  }
}

function prevSlide() {
  if (counter <= 0) {
    counter = carouselItems.length - 1;
    moveCarousel(true);
    isAutoPlay = true;
  } else {
    counter--;
    moveCarousel(false);
  }
}

nextBtn.addEventListener('click', () => {
  nextSlide();
  isAutoPlay = false;
});

prevBtn.addEventListener('click', () => {
  prevSlide();
  isAutoPlay = false;
});

setInterval(() => {
  if (isAutoPlay) {
    nextSlide();
  }
}, 10000);

window.addEventListener('resize', () => {
  moveCarousel(true);
});

// ---- Read more / Read less ----
document.querySelectorAll('.carousel-item p').forEach((p) => {
  p.classList.add('bio-text');

  const btn = document.createElement('button');
  btn.type = 'button';
  btn.className = 'read-more-btn';
  btn.textContent = 'Read more';

  btn.addEventListener('click', () => {
    const expanded = p.classList.toggle('expanded');
    btn.textContent = expanded ? 'Read less' : 'Read more';
  });

  p.insertAdjacentElement('afterend', btn);
});
</script>
